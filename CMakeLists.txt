cmake_minimum_required(VERSION 3.25)
cmake_policy(SET CMP0141 NEW)
project(OptAgentNative VERSION 0.1.0 LANGUAGES C CXX)
include(GNUInstallDirs)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_INTERPROCEDURAL_OPTIMIZATION OFF)
set(CMAKE_INSTALL_LIBDIR lib CACHE STRING "SDK library directory" FORCE)
set(OPTAGENT_NATIVE_LINKAGE STATIC CACHE STRING "STATIC or SHARED SDK")
set(OPTAGENT_SDK_COMPONENT common CACHE STRING "SDK module to produce")
set(OPTAGENT_SDK_SANITIZER "" CACHE STRING "address or thread instrumentation")
option(OPTAGENT_SDK_HIPO "Build the HiPO capability" ON)
if(OPTAGENT_NATIVE_LINKAGE STREQUAL "SHARED")
  set(BUILD_SHARED_LIBS ON CACHE BOOL "Shared common runtime" FORCE)
elseif(OPTAGENT_NATIVE_LINKAGE STREQUAL "STATIC")
  set(BUILD_SHARED_LIBS OFF CACHE BOOL "Static common runtime" FORCE)
else()
  message(FATAL_ERROR "Unsupported SDK linkage: ${OPTAGENT_NATIVE_LINKAGE}")
endif()
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
  set(CMAKE_MSVC_DEBUG_INFORMATION_FORMAT Embedded)
  add_compile_definitions(NOMINMAX WIN32_LEAN_AND_MEAN)
endif()
if(OPTAGENT_SDK_SANITIZER)
  if(MSVC OR NOT OPTAGENT_SDK_SANITIZER MATCHES "^(address|thread)$")
    message(FATAL_ERROR "Unsupported sanitizer/toolchain combination")
  endif()
  add_compile_options("-fsanitize=${OPTAGENT_SDK_SANITIZER}" -fno-omit-frame-pointer)
  add_link_options("-fsanitize=${OPTAGENT_SDK_SANITIZER}")
endif()
if(APPLE)
  set(CMAKE_INSTALL_RPATH "@loader_path")
elseif(UNIX)
  set(CMAKE_INSTALL_RPATH "$ORIGIN")
endif()
if(OPTAGENT_SDK_COMPONENT STREQUAL "probe")
  include("${CMAKE_CURRENT_SOURCE_DIR}/cmake/identity.cmake")
  sdk_toolchain_identity(_identity)
  file(WRITE "${CMAKE_BINARY_DIR}/identity.txt" "${_identity}\n")
  file(WRITE "${CMAKE_BINARY_DIR}/toolchain.json"
    "{\"compiler_id\":\"${CMAKE_CXX_COMPILER_ID}\",\"compiler_version\":\"${CMAKE_CXX_COMPILER_VERSION}\",\"system\":\"${CMAKE_SYSTEM_NAME}\",\"processor\":\"${CMAKE_SYSTEM_PROCESSOR}\",\"cxx_standard\":20}\n")
  return()
endif()
if(NOT EXISTS "${OPTAGENT_SDK_SOURCES}")
  message(FATAL_ERROR "Prepare the pinned source manifest with sdk.py first")
endif()
include("${OPTAGENT_SDK_SOURCES}")
if(NOT OPTAGENT_SDK_COMPONENT MATCHES "^(host|common|support|highs|cpsat|test-tools)$")
  message(FATAL_ERROR "Unknown SDK component: ${OPTAGENT_SDK_COMPONENT}")
endif()
include("${CMAKE_CURRENT_SOURCE_DIR}/cmake/${OPTAGENT_SDK_COMPONENT}.cmake")
file(WRITE "${CMAKE_BINARY_DIR}/toolchain.json"
  "{\"compiler_id\":\"${CMAKE_CXX_COMPILER_ID}\",\"compiler_version\":\"${CMAKE_CXX_COMPILER_VERSION}\",\"system\":\"${CMAKE_SYSTEM_NAME}\",\"processor\":\"${CMAKE_SYSTEM_PROCESSOR}\",\"cxx_standard\":20,\"build_type\":\"${CMAKE_BUILD_TYPE}\",\"linkage\":\"${OPTAGENT_NATIVE_LINKAGE}\",\"sanitizer\":\"${OPTAGENT_SDK_SANITIZER}\"}\n")
